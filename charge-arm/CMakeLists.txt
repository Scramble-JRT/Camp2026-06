cmake_minimum_required(VERSION 3.13)
project(charge_arm CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

find_package(OpenCV REQUIRED)
find_package(Threads REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(TESS REQUIRED tesseract lept)

add_executable(charge_arm main.cpp)
target_include_directories(charge_arm PRIVATE ${OpenCV_INCLUDE_DIRS} ${TESS_INCLUDE_DIRS})
target_link_directories(charge_arm PRIVATE ${TESS_LIBRARY_DIRS})
target_link_libraries(charge_arm PRIVATE ${OpenCV_LIBS} ${TESS_LIBRARIES} Threads::Threads)
